(defun templates--ensure-string (value name)
  "Signal an error unless VALUE is a string.
NAME is used in the error message."
  (unless (stringp value)
    (signal 'wrong-type-argument
            (list 'stringp (list name value))))
  value)


(defun templates--ensure-list (value name)
  "Signal an error unless VALUE is a list.
NAME is used in the error message."
  (unless (listp value)
    (signal 'wrong-type-argument
            (list 'listp (list name value))))
  value)


(defun string-pad-right (str total-len)
  "Pad STR on the right with spaces until it reaches TOTAL-LEN.

If STR is already at least TOTAL-LEN characters long, return it
unchanged."
  (templates--ensure-string str 'str)

  (unless (integerp total-len)
    (signal 'wrong-type-argument
            (list 'integerp total-len)))

  (when (< total-len 0)
    (error "TOTAL-LEN must not be negative: %d" total-len))

  (let ((length (length str)))
    (if (>= length total-len)
        str
      (concat str
              (make-string (- total-len length) ?\s)))))

(defun templates--ensure-nonempty-string (value name)
  "Signal an error unless VALUE is a non-empty string.

NAME identifies VALUE in the resulting error message."
  (templates--ensure-string value name)
  (when (string-empty-p value)
    (user-error "%s must not be empty" name))
  value)

(defun templates--join-nonempty (strings separator)
  "Join non-empty STRINGS using SEPARATOR."
  (mapconcat #'identity
             (seq-filter
              (lambda (string)
                (and (stringp string)
                     (not (string-empty-p string))))
              strings)
             separator))

(defun templates--to-string (object)
  "Convert OBJECT to Org syntax.

Strings are treated as plain text.  Org-element AST nodes are
interpreted with `org-element-interpret-data'.  Template structures
defined in this file provide their own `*-to-string' functions."
  (cond
   ((stringp object)
    object)

   ((and (listp object)
         (symbolp (car object))
         (memq (car object)
               (append org-element-all-elements
                       org-element-all-objects
                       '(org-data))))
    (org-element-interpret-data object))

   ((fboundp
     (intern (format "%s-to-string"
                     (type-of object))))
    (funcall
     (intern (format "%s-to-string"
                     (type-of object)))
     object))

   (t
    (user-error "Cannot serialize object: %S" object))))

;;; --------------------------------------------------------------------------
;;; Link
;;; --------------------------------------------------------------------------
(cl-defstruct
    (link
     (:constructor make-link
                   (&key type path
                         raw-link
                         format
                         description
                         search-option
                         application
                         explicit-type-p)))
  "A link object.

Corresponds to the `link' object in `org-element'.

TYPE is the link type, such as \"file\", \"https\", \"id\",
\"custom-id\" or \"fuzzy\".

PATH is the destination after the explicit link type has been
removed.

RAW-LINK is the original link target.

FORMAT is one of `plain', `angle' or `bracket'.

DESCRIPTION is the contents of a bracket link.  It is a list
because links are recursive objects.

SEARCH-OPTION and APPLICATION correspond to the respective
`org-element' properties.

EXPLICIT-TYPE-P records whether the link explicitly specified
its type."
  type
  path
  raw-link
  format
  description
  search-option
  application
  explicit-type-p)


(defun link-to-string (link)
  "Serialize LINK to Org syntax."
  (let* ((format (or (link-format link) 'bracket))
         (raw-link (or (link-raw-link link)
                       (link-path link)))
         (description (link-description link)))
    (pcase format
      ('plain
       raw-link)

      ('angle
       (format "<%s>" raw-link))

      ('bracket
       (if description
           (format "[[%s][%s]]"
                   raw-link
                   (templates--join-lines description))
         (format "[[%s]]" raw-link)))

      (_
       (user-error "Invalid link format: %S" format)))))


(defun make-simple-link (path &optional description)
  "Create a bracket LINK to PATH.

DESCRIPTION, when non-nil, is the link description."
  (make-link
   :type nil
   :path path
   :raw-link path
   :format 'bracket
   :description
   (when description
     (if (listp description)
         description
       (list description)))))

                    (item-checkbox item)))
         (tag (item-tag-to-string
               (item-tag item)))
         (contents (item-contents item))
         (prefix
          (concat bullet
                  (when checkbox
                    (concat " " checkbox))
                  (when tag
                    (concat " " tag
                            " ::")))))
    (if contents
        (concat
         prefix
         " "
         (templates--join-lines contents))
      prefix)))

;;; --------------------------------------------------------------------------
;;; Keyword
;;; --------------------------------------------------------------------------

;; `keyword' is a real `org-element' type.
;;
;; org-element representation:
;;
;;   (keyword (:key "TITLE" :value "..." ...))
;;
;; The two semantic fields are therefore KEY and VALUE.

(cl-defstruct (keyword
               (:constructor create-keyword (key value)))
  "A KEYWORD represents an Org `keyword' element."
  (key   nil :type string)
  (value nil :type string))


(defun keyword-to-string (keyword)
  "Serialize KEYWORD using Org keyword syntax.

The returned string has no trailing newline."
  (unless (keyword-p keyword)
    (signal 'wrong-type-argument
            (list 'keyword-p keyword)))

  (let ((key   (keyword-key keyword))
        (value (keyword-value keyword)))

    (templates--ensure-string key 'key)
    (templates--ensure-string value 'value)

    (when (string-match-p "[\n\r]" key)
      (error "Keyword key must be a single line: %S" key))

    (when (string-match-p "[\n\r]" value)
      (error "Keyword value must be a single line: %S" value))

    ;; Org keyword names are conventionally represented in upper case by
    ;; org-element, while Org itself accepts case-insensitive names.
    (format "#+%s: %s"
            (upcase key)
            value)))

;;; --------------------------------------------------------------------------
;;; Timestamp
;;; --------------------------------------------------------------------------

;; `timestamp' is an Org OBJECT, not an element.
;;
;; Its org-element representation has a number of derived properties:
;;
;;   :type
;;   :raw-value
;;   :year-start
;;   :month-start
;;   :day-start
;;   ...
;;
;; We deliberately store only TYPE and RAW-VALUE here.
;;
;; This is preferable to the previous START/END representation because
;; START/END are not the representation used by org-element and would
;; force this template layer to duplicate Org's timestamp parser.
;;
;; TYPE is one of:
;;
;;   active
;;   inactive
;;   active-range
;;   inactive-range
;;   diary
;;
;; RAW-VALUE is the complete timestamp syntax, including delimiters.

(defconst templates--timestamp-types
  '(active inactive active-range inactive-range diary)
  "Timestamp types recognized by `org-element'.")


(cl-defstruct (timestamp
               (:constructor create-timestamp (type raw-value)))
  "A TIMESTAMP represents an Org `timestamp' object."
  (type      nil :type symbol)
  (raw-value nil :type string))


(defun timestamp-type-p (type)
  "Return non-nil when TYPE is a valid Org timestamp type."
  (memq type templates--timestamp-types))


(defun timestamp-to-string (timestamp)
  "Serialize TIMESTAMP to its original Org syntax.

RAW-VALUE is preserved rather than reconstructed.  This mirrors
org-element's `:raw-value' property and avoids losing information
such as ranges, times, repeaters, warnings or diary syntax."
  (unless (timestamp-p timestamp)
    (signal 'wrong-type-argument
            (list 'timestamp-p timestamp)))

  (let ((type      (timestamp-type timestamp))
        (raw-value (timestamp-raw-value timestamp)))

    (unless (timestamp-type-p type)
      (error "Invalid Org timestamp type: %S" type))

    (templates--ensure-string raw-value 'raw-value)

    (when (string-match-p "[\n\r]" raw-value)
      (error "Timestamp must be a single line: %S" raw-value))

    raw-value))


(defun templates--timestamp-type-at-string (value)
  "Return the `org-element' timestamp type parsed from VALUE.

Signal an error if VALUE is not a valid standalone Org timestamp."
  (with-temp-buffer
    (insert value "\n")
    (goto-char (point-min))

    (let ((element (org-element-context)))
      (unless (eq (org-element-type element) 'timestamp)
        (error "Invalid Org timestamp: %S" value))
      (org-element-property :type element))))


(defun create-timestamp-from-string (value)
  "Create a TIMESTAMP from an Org timestamp string VALUE.

The timestamp type is obtained from `org-element', so this constructor
does not duplicate Org's timestamp grammar."
  (templates--ensure-string value 'value)

  (let ((type (templates--timestamp-type-at-string value)))
    (create-timestamp type value)))

* /headings/
Em ~org-mode~, definimos secções de um documento através de /headings/. As /headings/ são prefixadas por asteriscos ~*~, seguidas de algumas palavras-chave (e.g. TODO /keywords/), por um *título* e por uma *lista de /tags/.*

#+begin_src emacs-lisp
;;; --------------------------------------------------------------------------
;;; Heading
;;; --------------------------------------------------------------------------

;; IMPORTANT:
;;
;; `heading' is NOT an `org-element' type.
;;
;; An actual `headline' stores these concepts as properties such as:
;;
;;   :level
;;   :todo-keyword
;;   :raw-value
;;   :title
;;   :tags
;;
;; We retain this abstraction because it is useful for constructing a
;; headline, but it must not be confused with an AST node.

(cl-defstruct (heading
               (:constructor create-heading (level keyword title tags)))
  "A template abstraction for the properties of an Org headline."
  (level   1    :type integer)
  (keyword nil  :type (or null string))
  (title   nil  :type string)
  (tags    nil  :type list))


(defun level-to-stars (level)
  "Return LEVEL asterisks.

LEVEL must be a positive integer."
  (unless (integerp level)
    (signal 'wrong-type-argument
            (list 'integerp level)))

  (when (< level 1)
    (error "Org headline level must be at least 1: %d" level))

  (make-string level ?*))
#+end_src

** ~heading~ \to ~string~
O método abaixo transforma uma /struct/ ~Heading~ em uma /string/ com o formato ~*** KEYWORD título :tag1:tag2:~, em que a /keyword/ e as /tags/ são opcionais. Como ~drawer-to-string~, o resultado /não/ termina em uma quebra de linha.

O título não pode ser vazio: o Org não reconhece ~* TODO~ sem título como uma /keyword/ (lê ~TODO~ como o título) e ~*~ sozinho nem é um /heading/. Pela mesma razão, uma /keyword/ não pode conter espaços.

#+name:    templates--documents--heading2string 
#+date:    <2026-09-26>
#+version: 1.0.1
#+begin_src emacs-lisp
(defun heading-to-string (heading)
  "Serialize HEADING as an Org headline line.

This abstraction represents the following subset of headline syntax:

  STARS [TODO] TITLE [TAGS]

Priority cookies, comments and other headline features are deliberately
not represented by this structure yet."
  (unless (heading-p heading)
    (signal 'wrong-type-argument
            (list 'heading-p heading)))

  (let ((level   (heading-level heading))
        (keyword (heading-keyword heading))
        (title   (heading-title heading))
        (tags    (heading-tags heading)))

    (unless (stringp title)
      (signal 'wrong-type-argument
              (list 'stringp title)))

    (when (string-match-p "\\`[[:space:]]*\\'" title)
      (error "Headline title must not be empty"))

    (when (string-match-p "[\n\r]" title)
      (error "Headline title must be a single line: %S" title))

    (when (and keyword
               (or (not (stringp keyword))
                   (string-empty-p keyword)
                   (string-match-p "[[:space:]]" keyword)))
      (error "Invalid TODO keyword: %S" keyword))

    (let ((stars (level-to-stars level))
          (tag-string (templates--tags-to-string tags)))
      (concat
       stars
       " "
       (when keyword
         (concat keyword " "))
       title
       (when (not (string-empty-p tag-string))
         (concat " " tag-string))))))
#+end_src

;;; --------------------------------------------------------------------------
;;; Generic Drawer
;;; --------------------------------------------------------------------------

;; `drawer' is a real greater element.
;;
;; Its semantic property in org-element is :drawer-name.  The contents are
;; themselves elements.  The old representation as a list of arbitrary
;; key/value pairs was therefore too restrictive for a generic drawer.
;;
;; We keep PROPERTIES as a compatibility-oriented convenience only for now,
;; but validate that it contains node-property objects.  A generic drawer
;; serializer can still emit arbitrary textual contents through CONTENTS.

(cl-defstruct (drawer
               (:constructor create-drawer (name contents)))
  "A DRAWER represents a generic Org `drawer' greater element."
  (name     nil :type string)
  (contents nil :type list))


(defun templates--drawer-header (name)
  "Return the opening line for drawer NAME."
  (templates--ensure-string name 'name)

  (when (or (string-empty-p name)
            (string-match-p "[\n\r:]" name))
    (error "Invalid Org drawer name: %S" name))

  (format ":%s:" (upcase name)))


(defun templates--drawer-end ()
  "Return the structural closing line of an Org drawer."
  ":END:")


(defun templates--drawer-content-to-string (content)
  "Serialize one CONTENT item belonging to a drawer."
  (cond
   ((stringp content)
    (string-trim-right content))

   ((node-property-p content)
    (node-property-to-string content))

   ((key-value-pair-p content)
    ;; Compatibility with the old representation.
    (format ":%s: %s"
            (upcase (key-value-pair-key content))
            (key-value-pair-value content)))

   (t
    (error "Unsupported drawer content: %S" content))))


(defun drawer-to-string (drawer)
  "Serialize DRAWER as an Org drawer."
  (unless (drawer-p drawer)
    (signal 'wrong-type-argument
            (list 'drawer-p drawer)))

  (let ((name     (drawer-name drawer))
        (contents (drawer-contents drawer)))

    (templates--ensure-string name 'name)
    (templates--ensure-list contents 'contents)

    (mapconcat
     #'identity
     (append
      (list (templates--drawer-header name))
      (mapcar #'templates--drawer-content-to-string contents)
      (list (templates--drawer-end)))
     "\n")))

* documentos Org: alto nível
Os registros abaixo tentam incorporar toda a sintaxe Org, alinhados aos tipos que o próprio ~org-element~ usa: um documento é uma lista de /headlines/, e cada /headline/ tem, opcionalmente, uma /section/ — seu conteúdo imediato, antes de qualquer /headline/ filha.

** ~Section~
Uma /secção/ (~Section~), no sentido do ~org-element~, é *só* o conteúdo imediato sob uma /headline/ — texto, /drawers/, o que vier — sem a própria /heading/ e sem as /headlines/ filhas. É a contrapartida do tipo ~section~ que ~org-element-parse-buffer~ devolve.

#+name:    templates--documents--SectionType 
#+date:    <2026-09-26>
#+version: 1.0.1
#+begin_src emacs-lisp
;;; --------------------------------------------------------------------------
;;; Section
;;; --------------------------------------------------------------------------

;; `section' is a GREATER ELEMENT.
;;
;; It is NOT just a string.  It contains elements/objects representing
;; everything in the section before the next headline.
;;
;; We therefore store CONTENTS as a list.
;;
;; Strings are permitted as raw/plain text because the parser layer has not
;; yet introduced structs for paragraph, plain-list, table, etc.

(cl-defstruct (section
               (:constructor create-section (contents)))
  "A SECTION represents an Org `section' greater element."
  (contents nil :type list))


(defun templates--element-to-string (element)
  "Serialize one supported template ELEMENT."
  (cond
   ((stringp element)
    element)

   ((keyword-p element)
    (keyword-to-string element))

   ((timestamp-p element)
    (timestamp-to-string element))

   ((drawer-p element)
    (drawer-to-string element))

   ((property-drawer-p element)
    (property-drawer-to-string element))

   ((node-property-p element)
    (node-property-to-string element))

   ((heading-p element)
    (heading-to-string element))

   ((headline-p element)
    (headline-to-string element))

   ((section-p element)
    (section-to-string element))

   ((document-p element)
    (document-to-string element))

   ;; Permit an actual org-element AST node.  This is useful during the
   ;; transition between the parser and this template representation.
   ((and (listp element)
         (symbolp (car element))
         (memq (car element)
               (append org-element-all-elements
                       org-element-all-objects)))
    (org-element-interpret-data element))

   (t
    (error "Unsupported Org template element: %S" element))))

** ~Headline~
Uma /headline/ completa do documento — /heading/ + /section/ imediata + /headlines/ filhas — é o que o ~org-element~ chama de ~headline~. (Isto era chamado ~Section~ numa versão anterior deste documento; renomeado para bater com o tipo real do ~org-element~, já que ali "section" tem um sentido mais restrito — ver acima.)

#+name:    templates--documents--HeadlineType 
#+date:    <2026-09-26>
#+version: 1.0.0
#+begin_src emacs-lisp
;;; --------------------------------------------------------------------------
;;; Headline
;;; --------------------------------------------------------------------------

;; An `org-element' headline is itself the AST node.
;;
;; It contains a section and nested headlines as contents:
;;
;;   headline
;;     section
;;     headline
;;     headline
;;
;; There is no separate `heading' child in org-element.
;;
;; The `heading' field here remains a template abstraction for the headline
;; line itself.

(cl-defstruct (headline
               (:constructor create-headline (heading section children)))
  "A HEADLINE represents an Org `headline' greater element."
  (heading  nil :type heading)
  (section  nil :type (or null section))
  (children nil :type list))


(defun headline-to-string (headline)
  "Serialize HEADLINE recursively."
  (unless (headline-p headline)
    (signal 'wrong-type-argument
            (list 'headline-p headline)))

  (let* ((heading  (headline-heading headline))
         (section  (headline-section headline))
         (children (headline-children headline))
         (parent-level (heading-level heading))
         (parts '()))

    (unless (heading-p heading)
      (signal 'wrong-type-argument
              (list 'heading-p heading)))

    (unless (or (null section)
                (section-p section))
      (signal 'wrong-type-argument
              (list 'section-p section)))

    (templates--ensure-list children 'children)

    (push (heading-to-string heading) parts)

    (when section
      (let ((text (section-to-string section)))
        (unless (string-empty-p text)
          (push text parts))))

    (dolist (child children)
      (unless (headline-p child)
        (signal 'wrong-type-argument
                (list 'headline-p child)))

      (let ((child-level
             (heading-level
              (headline-heading child))))

        (unless (> child-level parent-level)
          (error
           "Child headline level (%d) must be greater than parent level (%d)"
           child-level
           parent-level)))

      (push (headline-to-string child) parts))

    (mapconcat #'identity
               (nreverse parts)
               "\n\n")))
#+end_src

*** ~headline~ \to ~string~
O método abaixo transforma uma /struct/ ~Headline~ em uma /string/, recursivamente: primeiro a /heading/ (via ~heading-to-string~), depois a /section/ imediata — se houver e não for vazia —, depois cada /headline/ filha, nessa ordem, cada bloco separado por uma linha em branco. Como ~heading-to-string~, o resultado /não/ termina em uma quebra de linha.

Uma /headline/ filha só é de fato uma /subheadline/ se seu /heading/ tiver nível maior que o da /headline/ pai — do contrário, o Org a leria como uma /headline/ irmã, e a árvore que ~Headline~ representa não corresponderia ao texto gerado. Por isso, ~headline-to-string~ verifica o nível de cada filha antes de gerar a /string/.

#+name:    templates--documents--headline2string 
#+date:    <2026-09-26>
#+version: 1.0.0
#+begin_src emacs-lisp
(defun headline-to-string (headline)
  "Transforms a HEADLINE to an `org-mode' structured string.
Recursively includes HEADLINE's own heading, its immediate section
and every child headline, in that order.  Signal an error if a
child's heading level is not greater than HEADLINE's own heading
level, since such a child would not actually nest under it once the
result is parsed."
  ;; type check validation
  (unless (headline-p headline)
    (signal 'wrong-type-argument (list 'headline-p headline)))

  ;; var. definition
  (let* ((heading  (headline-heading headline))
         (section  (headline-section headline))
         (children (headline-children headline))
         (lines    '()))

    ;; type check validation of the remaining fields
    (unless (heading-p heading)
      (signal 'wrong-type-argument (list 'heading-p heading)))
    (unless (or (null section) (section-p section))
      (signal 'wrong-type-argument (list 'section-p section)))
    (unless (listp children)
      (signal 'wrong-type-argument (list 'listp children)))
    (dolist (child children)
      (unless (headline-p child)
        (signal 'wrong-type-argument (list 'headline-p child)))
      (unless (> (heading-level (headline-heading child)) (heading-level heading))
        (error "Child headline level must be greater than its parent's: %S"
               (heading-level (headline-heading child)))))

    ;; (1.) add the heading
    (push (heading-to-string heading) lines)

    ;; (2.) add the immediate section, after a blank line, unless empty
    (when section
      (let ((s (section-to-string section)))
        (unless (string-empty-p s)
          (push "" lines)
          (push s lines))))

    ;; (3.) add each child headline, after a blank line
    (dolist (child children)
      (push "" lines)
      (push (headline-to-string child) lines))

    ;; (4.) reverse the lines and join them using `\n'
    (mapconcat #'identity (nreverse lines) "\n")))
#+end_src

;;; --------------------------------------------------------------------------
;;; Document / org-data
;;; --------------------------------------------------------------------------

;; `org-data' is the virtual root of an Org parse tree.
;;
;; It has no plist of its own and contains the document's top-level
;; contents.  At the greater-element level this normally consists of:
;;
;;   section
;;   headline
;;   headline
;;   ...
;;
;; We call our corresponding template abstraction `document'.

(cl-defstruct (document
               (:constructor create-document (section headlines)))
  "A DOCUMENT represents the virtual `org-data' root of an Org buffer."
  (section   nil :type (or null section))
  (headlines nil :type list))


(defun document-to-string (document)
  "Serialize DOCUMENT as a complete Org document."
  (unless (document-p document)
    (signal 'wrong-type-argument
            (list 'document-p document)))

  (let ((section   (document-section document))
        (headlines (document-headlines document))
        (parts     '()))

    (unless (or (null section)
                (section-p section))
      (signal 'wrong-type-argument
              (list 'section-p section)))

    (templates--ensure-list headlines 'headlines)

    ;; Top-level section comes before top-level headlines.
    (when section
      (let ((text (section-to-string section)))
        (unless (string-empty-p text)
          (push text parts))))

    (dolist (headline headlines)
      (unless (headline-p headline)
        (signal 'wrong-type-argument
                (list 'headline-p headline)))

      (push (headline-to-string headline) parts))

    (mapconcat #'identity
               (nreverse parts)
               "\n\n")))

#+end_
